// $'…' (ANSI-C quoting) is decoded before the path checks, so $'\x2e'claude or $'\154n' is read as written (Q-0418 (a)).
const ESC = { a: '\x07', b: '\b', e: '\x1b', E: '\x1b', f: '\f', n: '\n', r: '\r', t: '\t', v: '\v' };
const ansi = (s) => s.replace(/\\(x[0-9a-fA-F]{1,2}|u[0-9a-fA-F]{1,4}|U[0-9a-fA-F]{1,8}|[0-7]{1,3}|.)/g, (m, e) =>
  (/^[xuU]/.test(e) ? String.fromCodePoint(parseInt(e.slice(1), 16)) : /^[0-7]/.test(e) ? String.fromCharCode(parseInt(e, 8)) : ESC[e] ?? e));
const ANSI = /\$'((?:[^'\\]|\\.)*)'/g;
const plain = raw.replace(ANSI, (m, a) => `'${ansi(a).replace(/'/g, '')}'`);
