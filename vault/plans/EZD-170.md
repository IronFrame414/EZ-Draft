**Helper** (step 1; tested 2026-10-24: build EZD-034 session `4596eb80-5e84-4c6f-98c9-7c24e10ded51` → V0's value; build EZD-018 session `434ff708-4475-4967-ae50-105b2a9205a7` → `"unfinished":["builder"],"resumes":12`):
```
import { readFileSync } from 'node:fs';
const txt = readFileSync(process.argv[2], 'utf8');
const rows = txt.split('\n').filter(Boolean).map((l) => { try { return JSON.parse(l); } catch { return null; } }).filter(Boolean);
const uses = []; const sync = new Set(); let resumes = 0;
for (const r of rows) {
  const c = r.message?.content; if (!Array.isArray(c)) continue;
  for (const b of c) {
    if (b.type === 'tool_use' && b.name === 'Agent') uses.push({ id: b.id, type: b.input?.subagent_type, model: b.input?.model ?? null });
    if (b.type === 'tool_use' && b.name === 'SendMessage') resumes += 1;
    if (b.type === 'tool_result' && !JSON.stringify(b.content ?? '').includes('launched successfully')) sync.add(b.tool_use_id);
  }
}
const done = new Set([...txt.matchAll(/<tool-use-id>(toolu_\w+)<\\?\/tool-use-id>[\s\S]{0,400}?<status>completed<\\?\/status>/g)].map((m) => m[1]));
const unfinished = uses.filter((u) => !sync.has(u.id) && !done.has(u.id)).map((u) => u.type);
console.log(JSON.stringify({ agents: uses.length, unfinished, overrides: uses.filter((u) => u.model).map((u) => `${u.type}:${u.model}`), resumes }));
```
