D3 script (replace nothing; the worktree path is literal):
```bash
#!/bin/bash
# EZD-351 five-run check (vault/plans/EZD-351.md). Scratch file, never committed.
set -u
WT=/Users/josh/Projects/ezdraft/.claude/worktrees/EZD-351
T=$(node -p 'require("os").tmpdir()')
OUT="$T/ezd-351"; mkdir -p "$OUT"; : > "$OUT/summary.txt"
L1="$WT/.lanes/lane1"
say() { echo "$(date -u +%FT%TZ) $*" | tee -a "$OUT/summary.txt"; }
live() { p=$(tr -d '[:space:]' 2>/dev/null < "$1"); [[ "$p" =~ ^[1-9][0-9]*$ ]] && kill -0 "$p" 2>/dev/null; }
GATES=0; UP=0
finish() {
  code=$1; shift; say "$*"
  if [ $UP -eq 1 ]; then EZD_LANE=1 SUPABASE_WORKDIR="$L1" pnpm db:stop > "$OUT/lane1-db-stop.log" 2>&1; say "lane 1 stack stopped (exit $?)"; fi
  for s in $(seq 1 2700); do [ -z "$(jobs -rp)" ] && break; sleep 1; done
  if [ -n "$(jobs -rp)" ]; then say "TIMEOUT: lane 2 gate still running after 2700 s; it stops its own stack when it ends"; code=3; fi
  for g in "$OUT"/lane2-gate-*.log; do [ -f "$g" ] && say "$(basename "$g"): $(grep -E '^GATE (PASS|FAIL)' "$g" | tail -1) $(tail -1 "$g")"; done
  exit $code
}
for f in ezdraft-gate-lane1.lock ezdraft-gate-lane2.lock ezdraft-lane1.lease ezdraft-lane2.lease; do
  if live "$T/$f"; then say "BLOCKED: $f is held by a live process"; exit 2; fi
done
echo $$ > "$T/ezdraft-lane1.lease"
trap 'rm -f "$T/ezdraft-lane1.lease"' EXIT
start_gate() {
  if [ $GATES -ge 3 ]; then finish 2 "BLOCKED: the lane 2 gate ended before the runs finished, 3 times"; fi
  GATES=$((GATES+1))
  (cd "$WT" && EZD_LANE=2 node scripts/gate.mjs "$WT" > "$OUT/lane2-gate-$GATES.log" 2>&1; echo "exit $?" >> "$OUT/lane2-gate-$GATES.log") &
  say "lane 2 gate $GATES started"
  for i in $(seq 1 120); do live "$T/ezdraft-gate-lane2.lock" && return 0; sleep 1; done
  finish 2 "BLOCKED: lane 2 gate $GATES took no lock in 120 s"
}
cd "$WT"
UP=1
if ! EZD_LANE=1 SUPABASE_WORKDIR="$L1" pnpm db:up > "$OUT/lane1-db-up.log" 2>&1; then finish 2 "BLOCKED: lane 1 db:up failed"; fi
say "lane 1 values: $(EZD_LANE=1 SUPABASE_WORKDIR="$L1" node scripts/db/local.mjs lane)"
start_gate
PASSES=0; FAILS=0; N=0
while [ $PASSES -lt 5 ] && [ $FAILS -eq 0 ] && [ $N -lt 8 ]; do
  N=$((N+1))
  live "$T/ezdraft-gate-lane2.lock" || start_gate
  say "run $N start, lane 2 gate lock live"
  (cd app && EZD_LANE=1 SUPABASE_WORKDIR="$L1" pnpm exec vitest run src/server/db/rls.audit.test.ts src/server/jobs/handlers.test.ts -t 'T18 exemption and grant lists match the catalog|T19 every registered handler has a run-twice test' --reporter=verbose) > "$OUT/run-$N.log" 2>&1
  code=$?
  if live "$T/ezdraft-gate-lane2.lock"; then lock=live; else lock=gone; fi
  if [ $code -ne 0 ]; then FAILS=$((FAILS+1)); verdict=failed; elif [ $lock = live ]; then PASSES=$((PASSES+1)); verdict=counted; else verdict="not counted"; fi
  say "run $N exit $code, lane 2 gate lock $lock at end, $verdict;$(grep -E '^ +Tests ' "$OUT/run-$N.log" | tr -s ' ')"
done
finish 0 "RESULT: $PASSES counted passes, $FAILS failed runs"
```
