```js
// Segments split at ; & | outside quotes only; a backslash outside single quotes escapes the next character (EZD-154; EZD-000i K1).
function segments(s) {
  const out = [];
  let cur = '';
  let q = '';
  for (let i = 0; i < s.length; i++) {
    const ch = s[i];
    if (q === "'") { cur += ch; if (ch === "'") q = ''; continue; }
    if (ch === '\\' && i + 1 < s.length) { cur += ch + s[i + 1]; i++; continue; }
    if (q === '"') { cur += ch; if (ch === '"') q = ''; continue; }
    if (ch === '"' || ch === "'") { q = ch; cur += ch; continue; }
    if (ch === ';' || ch === '&' || ch === '|') { out.push(cur); cur = ''; continue; }
    cur += ch;
  }
  out.push(cur);
  return out;
}
// Quote marks and backslashes are dropped before matching, so a quoted or escaped option or file name is still seen (EZD-154).
const unquote = (s) => s.replace(/["'\\]/g, '');
for (const seg of segments(raw)) {
  const s = unquote(seg);
  if (/\bsecurity\s+find-(generic|internet)-password\b.*\s-(?=[A-Za-z]*[wg])[A-Za-z]+(\s|$)/.test(s)) deny(KEYCHAIN_WHY);
  if (/\bsecurity\s+dump-keychain\b.*\s-(?=[A-Za-z]*d)[A-Za-z]+(\s|$)/.test(s)) deny(KEYCHAIN_WHY);
}
// The test -f and [ -f ] exemption covers one path token: a quoted string, or unquoted text with no space, quote, ; & | < > ( ) or
// backtick, followed by a space or the end (test -f) or by ] (bracket form) (EZD-154; EZD-000i K3).
const noTests = unquote(raw
  .replace(/(^|\s)test\s+-f\s+(?:"[^"]*"|'[^']*'|[^\s;&|<>()`'"]+)(?=\s|$)/g, '$1')
  .replace(/\[\s+-f\s+(?:"[^"]*"|'[^']*'|[^\s;&|<>()`'"]+)\s+\]/g, ''));
```
