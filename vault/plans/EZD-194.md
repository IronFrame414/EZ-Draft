**Block F** (the fake `claude`, one line per row):

```
#!/bin/sh
if [ "$1" = "--help" ]; then echo "--max-turns --session-id --permission-prompts"; exit 0; fi
{
  if [ "${ANTHROPIC_API_KEY+set}" = set ]; then printf 'key=%s\n' "$ANTHROPIC_API_KEY"; else printf 'key=unset\n'; fi
  printf 'scrub=%s\n' "${CLAUDE_CODE_SUBPROCESS_ENV_SCRUB-unset}"
} > '<ROOT>/claude-env.txt'
cp '<ROOT>/logs/inflight.json' '<ROOT>/inflight-seen.json'
printf '%s\n' '{"type":"result","subtype":"success","is_error":false,"result":"fake unit done","session_id":"ezd194-fake-session","num_turns":1,"duration_ms":5,"total_cost_usd":0,"modelUsage":{}}'
```
