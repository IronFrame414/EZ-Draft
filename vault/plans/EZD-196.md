import { readFileSync, readdirSync, existsSync } from 'node:fs';
import path from 'node:path';
const [dir, description, probeFile] = process.argv.slice(2);
const metas = readdirSync(dir).filter((f) => f.endsWith('.meta.json')).map((f) => ({ f, m: JSON.parse(readFileSync(path.join(dir, f), 'utf8')) }));
const hit = metas.filter((x) => x.m.description === description);
if (hit.length !== 1) { console.log(JSON.stringify({ error: `found ${hit.length} subagents with that description` })); process.exit(1); }
const transcript = path.join(dir, hit[0].f.replace(/\.meta\.json$/, '.jsonl'));
const uses = []; const results = new Map(); let model = null;
for (const l of (existsSync(transcript) ? readFileSync(transcript, 'utf8') : '').split('\n')) {
  if (!l) continue; let r; try { r = JSON.parse(l); } catch { continue; }
  if (r.message?.role === 'assistant' && r.message.model && !model) model = r.message.model;
  const c = r.message?.content; if (!Array.isArray(c)) continue;
  for (const b of c) {
    if (b.type === 'tool_use' && ['Write', 'Edit', 'NotebookEdit'].includes(b.name) && !uses.some((u) => u.id === b.id)) uses.push({ id: b.id, tool: b.name, file_path: b.input?.file_path ?? b.input?.notebook_path ?? null });
    if (b.type === 'tool_result' && !results.has(b.tool_use_id)) results.set(b.tool_use_id, { is_error: b.is_error === true, text: typeof b.content === 'string' ? b.content : JSON.stringify(b.content) });
  }
}
console.log(JSON.stringify({ transcript, agentType: hit[0].m.agentType, model,
  calls: uses.map((u) => ({ tool: u.tool, file_path: u.file_path, is_error: results.get(u.id)?.is_error ?? null, result: results.get(u.id)?.text ?? null })), probeExists: existsSync(probeFile) }));
```

import { readFileSync } from 'node:fs';
import path from 'node:path';
const [tr, meta, target] = process.argv.slice(2);
const m = JSON.parse(readFileSync(meta, 'utf8'));
let use = null;
for (const l of readFileSync(tr, 'utf8').split('\n')) { if (!l) continue; let r; try { r = JSON.parse(l); } catch { continue; }
  const c = r.message?.content; if (!Array.isArray(c)) continue;
  for (const b of c) if (b.type === 'tool_use' && b.name === 'Write') { use = { b, r }; break; } if (use) break; }
if (!use) { console.error('no Write tool_use'); process.exit(1); }
const input = { ...use.b.input };
if (target) input.file_path = path.join(use.r.cwd, target);
const payload = { session_id: use.r.sessionId, transcript_path: tr, cwd: use.r.cwd, hook_event_name: 'PreToolUse', tool_name: 'Write',
  tool_input: input, tool_use_id: use.b.id, agent_id: m.agentId ?? use.r.agentId, agent_type: m.agentType };
process.stdout.write(JSON.stringify(payload));
```
